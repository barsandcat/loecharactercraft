New card data structure

    "CardId": {
      "Name": "Card name", # "Required",
      "DisplayName": "Display name", # "For the basic actions upgrades, if present overrides name",
      "Type": "Action/Reaction", # Default "Action"
      "Category": "Offensive/Defensive/Support", # Default "Offensive"
	  "Keywords": ["Keyword1", "Keyword2"], # Optional
      "Condition": "Condition description", # Optional
	  "Target": "Range instructions", # Optional
      "Roll": {
		  "DIV": true, # if absent - false, i.e. no DIV rol
		  "Mode": "Sum", # if ablest - "Highest"
		  "ATT": ["INT"], # If absent, means all stats
		  "Difficulty": 8, # if absent, means difficulty of the enemy [EN]
		  "Modifiers": [
			{ "Dice": -2, "Triggers": ["Trigger1"] },
			{ "Dice": 1, "Triggers": ["Trigger2"] } 
		  ], # Optional
		  "Successes": {
			  "3": "Best result",
			  "1": "Ok result",
			  "0": "Failure"
		  } # "Optional"
	  } # Optional
	      "Front": "Front instructions", # Optional
	      "Back": "Back instructions", # Optional
	      "Note": "Back instructions in blue", # Optional
	      "Warning": "Back instructions in red", # Optional
	      "Level": 1 # Only relevant for basic action cards, if absent = 1
	    },
	
	
Old roll format: "Roll ATT and ATT, DIV [Number/EN]; +1: Trigger1, Trigger2; -2: Trigger3"
